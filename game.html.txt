    <script>
        // --- نظام الصوت المدمج في المتصفح ---
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        
        function playFireSound() {
            if(audioCtx.state === 'suspended') audioCtx.resume();
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);
            osc.type = 'square';
            osc.frequency.setValueAtTime(300, audioCtx.currentTime);
            osc.frequency.exponentialRampToValueAtTime(50, audioCtx.currentTime + 0.3);
            gain.gain.setValueAtTime(0.1, audioCtx.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            osc.start();
            osc.stop(audioCtx.currentTime + 0.3);
        }

        function playHitSound() {
            if(audioCtx.state === 'suspended') audioCtx.resume();
            [523.25, 659.25, 783.99].forEach((freq, i) => {
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.type = 'sine';
                osc.frequency.value = freq;
                gain.gain.setValueAtTime(0.2, audioCtx.currentTime + i * 0.1);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + i * 0.1 + 0.5);
                osc.start(audioCtx.currentTime + i * 0.1);
                osc.stop(audioCtx.currentTime + i * 0.1 + 0.5);
            });
        }
        // -------------------------------------

        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const angleSlider = document.getElementById('angle');
        const speedSlider = document.getElementById('speed');
        const fireBtn = document.getElementById('fireBtn');
        const valX = document.getElementById('valX');
        const valY = document.getElementById('valY');
        const successMsg = document.getElementById('successMsg');

        // الثوابت الفيزيائية
        const g = 9.8; 
        const scale = 10; // كل 10 بكسل = 1 متر
        const startX = 50; 
        const groundY = canvas.height - 100;
        
        let projectile = null;
        let target = { x: 800, w: 80, h: 20 };
        let animationId = null;

        // تحديث الأرقام عند سحب المؤشر
        angleSlider.addEventListener('input', () => document.getElementById('angleValue').textContent = angleSlider.value);
        speedSlider.addEventListener('input', () => document.getElementById('speedValue').textContent = speedSlider.value);

        function drawGrid() {
            ctx.strokeStyle = '#ecf0f1';
            ctx.lineWidth = 1;
            
            // خطوط عمودية
            for(let i = startX; i < canvas.width; i += scale * 10) {
                ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, canvas.height); ctx.stroke();
                if(i > startX) {
                    ctx.fillStyle = '#95a5a6'; ctx.font = '14px Tahoma';
                    ctx.fillText(((i - startX)/scale) + 'm', i - 15, groundY + 25);
                }
            }

            // خطوط أفقية
            for(let i = groundY; i > 0; i -= scale * 10) {
                ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(canvas.width, i); ctx.stroke();
                if(i < groundY) {
                    ctx.fillStyle = '#95a5a6'; ctx.font = '14px Tahoma';
                    ctx.fillText(((groundY - i)/scale) + 'm', startX - 35, i + 5);
                }
            }
        }

        function drawScene() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            drawGrid();

            // رسم الأرض والهدف
            ctx.fillStyle = '#27ae60';
            ctx.fillRect(0, groundY, canvas.width, canvas.height - groundY);
            ctx.fillStyle = '#f1c40f';
            ctx.fillRect(target.x, groundY - target.h, target.w, target.h);
            
            // --- رسم المدفع (المحسن) ---
            const angleRad = angleSlider.value * Math.PI / 180;
            ctx.save();
            ctx.translate(startX, groundY);
            
            // الفوهة تتحرك مع الزاوية
            ctx.rotate(-angleRad);
            ctx.fillStyle = '#555';
            ctx.fillRect(0, -10, 60, 20); // طول الفوهة 60 وسمكها 20
            ctx.restore();
            
            // قاعدة المدفع والعجلة
            ctx.fillStyle = '#7f8c8d';
            ctx.beginPath(); ctx.arc(startX, groundY, 30, Math.PI, 0); ctx.fill();
            ctx.strokeStyle = '#2c3e50'; ctx.lineWidth = 3; ctx.stroke();
            
            ctx.fillStyle = '#2c3e50';
            ctx.beginPath(); ctx.arc(startX, groundY, 12, 0, Math.PI * 2); ctx.fill();
            ctx.fillStyle = '#bdc3c7';
            ctx.beginPath(); ctx.arc(startX, groundY, 4, 0, Math.PI * 2); ctx.fill();
            
            // رسم مسار القذيفة
            if (projectile && projectile.path.length > 0) {
                ctx.beginPath();
                ctx.moveTo(projectile.path[0].x, projectile.path[0].y);
                for(let p of projectile.path) ctx.lineTo(p.x, p.y);
                ctx.strokeStyle = '#e74c3c';
                ctx.lineWidth = 2;
                ctx.stroke();
            }

            // رسم القذيفة
            if (projectile && projectile.active) {
                ctx.beginPath();
                ctx.arc(projectile.currentX, projectile.currentY, 8, 0, Math.PI * 2);
                ctx.fillStyle = '#2c3e50';
                ctx.fill();
            }
        }

        function generateRandomTarget() {
            const randomMeters = Math.floor(Math.random() * 60) + 30; // بين 30 و 90 متر
            target.x = startX + (randomMeters * scale);
        }

        function updatePhysics() {
            if (!projectile || !projectile.active) return;

            projectile.time += 0.05;
            let t = projectile.time;

            let xMeters = projectile.v0 * Math.cos(projectile.angle) * t;
            let yMeters = projectile.v0 * Math.sin(projectile.angle) * t - 0.5 * g * t * t;

            valX.textContent = xMeters.toFixed(2);
            valY.textContent = Math.max(0, yMeters).toFixed(2);

            projectile.currentX = startX + (xMeters * scale);
            projectile.currentY = groundY - (yMeters * scale);
            projectile.path.push({x: projectile.currentX, y: projectile.currentY});

            // التحقق من الاصطدام
            if (projectile.currentY >= groundY && t > 0.1) {
                projectile.currentY = groundY;
                projectile.active = false;
                
                if (projectile.currentX >= target.x && projectile.currentX <= target.x + target.w) {
                    playHitSound();
                    successMsg.style.opacity = '1';
                    setTimeout(() => {
                        successMsg.style.opacity = '0';
                        generateRandomTarget();
                        drawScene();
                    }, 2000);
                }
            }
        }

        function gameLoop() {
            updatePhysics();
            drawScene();
            if (projectile && projectile.active) {
                animationId = requestAnimationFrame(gameLoop);
            }
        }

        function fire() {
            if(animationId) cancelAnimationFrame(animationId);
            playFireSound();
            
            valX.textContent = "0.00";
            valY.textContent = "0.00";
            successMsg.style.opacity = '0';

            projectile = {
                v0: parseFloat(speedSlider.value),
                angle: parseFloat(angleSlider.value) * Math.PI / 180,
                time: 0,
                currentX: startX,
                currentY: groundY,
                active: true,
                path: []
            };
            gameLoop();
        }

        fireBtn.addEventListener('click', fire);

        // دعم الإطلاق باستخدام مفتاح المسافة (Space)
        document.addEventListener('keydown', (e) => {
            if (e.code === 'Space') {
                e.preventDefault(); // لمنع نزول الصفحة لأسفل
                fire();
            }
            // التحكم بالأسهم
            if (e.code === 'ArrowUp' && angleSlider.value < 90) { angleSlider.value++; angleSlider.dispatchEvent(new Event('input')); drawScene();}
            if (e.code === 'ArrowDown' && angleSlider.value > 0) { angleSlider.value--; angleSlider.dispatchEvent(new Event('input')); drawScene();}
            if (e.code === 'ArrowRight' && speedSlider.value < 40) { speedSlider.value++; speedSlider.dispatchEvent(new Event('input')); }
            if (e.code === 'ArrowLeft' && speedSlider.value > 5) { speedSlider.value--; speedSlider.dispatchEvent(new Event('input')); }
        });

        // تهيئة اللعبة
        generateRandomTarget();
        drawScene();
    </script>
